import { test } from "@playwright/test";

test.describe("Chụp ảnh nghiệm thu trực quan Gói 04A (Trang Người dân)", () => {
  test("Chụp ảnh Desktop 1440x900", async ({ page }) => {
    await page.setViewportSize({ width: 1440, height: 900 });
    await page.goto("http://localhost:3000/services");
    await page.waitForLoadState("networkidle");

    // Chụp màn hình đầu tiên (viewport)
    await page.screenshot({
      path: "tests/screenshots/desktop-services-viewport.png",
      fullPage: false,
    });

    // Chụp toàn trang desktop
    await page.screenshot({
      path: "tests/screenshots/desktop-services-full.png",
      fullPage: true,
    });

    // Tương tác bấm phân nhánh
    await page.getByRole("button", { name: /Chưa có hợp đồng văn bản/i }).click();
    await page.waitForTimeout(300);

    // Chụp trạng thái sau khi phân nhánh
    await page.screenshot({
      path: "tests/screenshots/desktop-services-branched.png",
      fullPage: true,
    });
  });

  test("Chụp ảnh Mobile 375x812 (iPhone view)", async ({ page }) => {
    await page.setViewportSize({ width: 375, height: 812 });
    await page.goto("http://localhost:3000/services");
    await page.waitForLoadState("networkidle");

    // Chụp màn hình đầu tiên (viewport)
    await page.screenshot({
      path: "tests/screenshots/mobile-services-viewport.png",
      fullPage: false,
    });

    // Chụp toàn trang mobile
    await page.screenshot({
      path: "tests/screenshots/mobile-services-full.png",
      fullPage: true,
    });
  });
});
